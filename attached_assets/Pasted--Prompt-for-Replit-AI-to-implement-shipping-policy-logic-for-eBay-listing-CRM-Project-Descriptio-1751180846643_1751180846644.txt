# Example Python Function to Map Weight to Shipping Policy

def get_shipping_policy_id(weight_grams: int, shipping_policy_map: dict) -> str:
    for weight_range, policy_id in shipping_policy_map.items():
        min_weight, max_weight = weight_range
        if min_weight <= weight_grams <= max_weight:
            return policy_id
    return "default_policy_id"  # Fallback if no match

# Example Usage
SHIPPING_POLICY_MAP = {
    (0, 99): "policy_light",
    (100, 499): "policy_medium",
    (500, 999): "policy_heavy",
    (1000, 4999): "policy_very_heavy",
    (5000, 9999): "policy_oversized",
}

product_weight = 450  # in grams
policy_id = get_shipping_policy_id(product_weight, SHIPPING_POLICY_MAP)
print(f"Use shipping policy: {policy_id}")
